# 11 DFS和BFS

## DFS 深度优先（类似先序遍历）

bool visited[MVNum];          // 1. 全局小本本：记录每个顶点的生死状态（是否被访问过）
void DFSTraverse(Graph G)
{
    for(v = 0; v < G.vexnum; ++v)
        visited[v] = false;   // 2. 初始化：开局把所有顶点都标记为“没去过”

    for(v = 0; v < G.vexnum; ++v) // 3. 挨个巡逻全图所有顶点
        if(!visited[v])       // 4. 只要发现有个顶点 v 还没被访问过（发现新大陆/新岛屿）
            DFS(G,v);         // 5. 立刻派敢死队从 v 点空降，把它所在的整块陆地全部探完
}


void DFS(Graph G, int v)
{
    cout << v;                // 1. 落地打卡：来到当前顶点 v，直接打印/访问它
    visited[v] = true;        // 2. 盖章标记：在小本本上把 v 勾选为“已访问”，防止等会儿走回头路

				    // 3. 核心大循环：寻找当前顶点 v 的所有邻居
				    // FirstAdjVex(G,v)：找到 v 的第一个邻居 w
				    // w >= 0：只要 w 存在（如果不存在邻居，这个函数通常返回 -1）
				    // NextAdjVex(G,v,w)：找 v 紧接着 w 的下一个邻居
    for(w = FirstAdjVex(G,v); w >= 0; w = NextAdjVex(G,v,w))  
    { 
        if(!visited[w])       // 4. 看看邻居 w 去过没有？
        {
            DFS(G,w);         // 5. 没去过？兄弟们冲！立刻套娃调用 DFS(w)，直接瞬移到 w 点继续往深处挖！
        }
    }
}

## BFS 广度优先（类似层次遍历）

bool visited[MVNum];          // 1. 全局小本本：记录每个顶点的生死状态（是否被访问过）
void BFSTraverse(Graph G)
{
    for(v = 0; v < G.vexnum; ++v)
        visited[v] = false;   // 2. 初始化：开局把所有顶点都标记为“没去过”

    for(v = 0; v < G.vexnum; ++v) // 3. 挨个巡逻全图所有顶点
        if(!visited[v])       // 4. 只要发现有个顶点 v 还没被访问过（发现新大陆/新岛屿）
            BFS(G,v);         // 5. 立刻派敢死队从 v 点空降，把它所在的整块陆地全部探完
}

void BFS(Graph G, int v)  
{
    cout << v;                // 1. 落地打卡：访问起始点 v
    visited[v] = true;        // 2. 盖章标记：标记 v 已经来过了
    InitQueue(Q);             // 3. 搬出一根传送带：初始化队列 Q
    EnQueue(Q, v);            // 4. 起始点上车：把起点 v 扔进==队列==

    while(!QueueEmpty(Q))     // 5. 核心大循环：只要传送带上还有人，搜救就不停
    {
        DeQueue(Q, u);        // 6. 队长下车：从队头拉出一个顶点 u，现在由 u 当临时指挥官
        
        // 7. 疯狂扫射邻居：遍历当前指挥官 u 的所有邻居 w
        for(w = FirstAdjVex(G,u); w >= 0; w = NextAdjVex(G,u,w))
        {
            if(!visited[w])   // 8. 发现有个邻居 w 还没被去过！
            {
                cout << w;            // 9. 隔空打卡：大喊一声 w 的名字（输出）
                visited[w] = true;    // 10. 遥控盖章：立刻把 w 标记为“已访问”（非常关键！）
                EnQueue(Q, w);        // 11. 邻居上车：把 w 塞进队尾排队，等下一轮轮到它当指挥官
            }
        }
    }
}
